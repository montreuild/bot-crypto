# ================================================================
#  Crypto Bot V12 - Python Dependencies
#  !! Python 3.14 REQUIRED !!
#
#  Installation on Ubuntu 22.04 (Python 3.10 by default):
#    sudo add-apt-repository ppa:deadsnakes/ppa -y
#    sudo apt install -y python3.14 python3.14-venv python3.14-dev
#    python3.14 -m venv /opt/crypto_bot/.venv
#    source /opt/crypto_bot/.venv/bin/activate
#    pip install -r requirements.txt
#
#  On Ubuntu 24.04+ (Python 3.12 native, 3.14 via deadsnakes):
#    sudo add-apt-repository ppa:deadsnakes/ppa -y
#    sudo apt update && sudo apt install -y python3.14 python3.14-venv python3.14-dev
#    python3.14 -m venv /opt/crypto_bot/.venv
#    source /opt/crypto_bot/.venv/bin/activate
#    pip install -r requirements.txt
#
#  On macOS (Homebrew):
#    brew install python@3.14
#    python3.14 -m venv .venv
#    source .venv/bin/activate
#    pip install -r requirements.txt
#
#  On Windows (Git Bash or PowerShell):
#    See docs/DEMARRAGE_WINDOWS.md for full guide.
#    1. Install Python 3.14 from https://www.python.org/downloads/
#       (check "Add Python to PATH")
#    2. py -3.14 -m venv .venv
#    3. source .venv/Scripts/activate   # Git Bash
#       OR  .venv\Scripts\Activate.ps1  # PowerShell
#    4. pip install -r requirements.txt
#
#  NOTE : les versions sont épinglées pour garantir la reproductibilité
#  en production. Pour mettre à jour, testez d'abord dans un environnement
#  de développement, puis mettez à jour les versions ici.
#
#  Date de lock : 2026-07-29 — vérifié par `pip install -r requirements.txt`
#  dans un venv Python 3.14 vierge (aucun conflit de résolution) suivi de
#  `pytest -q` (576 tests, tous verts). Toutes les dépendances directes sont
#  épinglées en version exacte (==) ; les sous-dépendances transitives ne le
#  sont pas (pas de lockfile — cf. TEST-12 dans docs/audit/07-tests-ci-docs.md
#  pour la justification de ce choix).
#
#  ⚠ Python 3.14 migration: free-threaded mode (PEP 779) is now stable.
#  LightGBM, Polars, Optuna, CCXT all support 3.14 since Q2 2026.
#  Free-threaded build (python3.14t) optional, see docs/DEMARRAGE_WINDOWS.md.
# ================================================================

# -- Web / API -------------------------------------------------------
fastapi==0.115.0
# starlette : transitive de fastapi, épinglée ICI malgré la règle « seules les
# dépendances directes sont épinglées » (cf. en-tête). La règle suppose que les
# transitives sont contraintes par leur parent — ce n'est vrai que d'un côté :
# fastapi 0.115.0 demande `starlette>=0.37.2,<0.39.0`, mais rien n'empêche une
# réinstallation de résoudre vers une version incompatible si un AUTRE paquet
# la tire. C'est arrivé : l'environnement de dev s'est retrouvé en starlette
# 1.3.1, et `APIRouter()` levait `TypeError: Router.__init__() got an
# unexpected keyword argument 'on_startup'` — toute la collecte des tests API
# échouait, sur un composant que personne n'avait touché.
starlette==0.38.6
uvicorn[standard]==0.30.0
# SEC-010 : 3.1.4 est vulnérable (CVE-2024-56201 / CVE-2024-56326, contournement
# du bac à sable de template). Le plancher de l'audit était 3.1.5 — on prend 3.1.6
# parce que 3.1.5 est elle-même vulnérable à CVE-2025-27516 (même famille) : viser
# le plancher exact aurait imposé un second bump immédiat.
jinja2==3.1.6
python-multipart==0.0.22
slowapi==0.1.9

# -- Multi-actifs / actions (G2) -------------------------------------

# ⚠ RÉINSTALLE PANDAS, supprimé du projet en phase 6 (~23 Mo avec ses
